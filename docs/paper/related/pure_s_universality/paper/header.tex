% Preserve useful link descriptions independently of the Hyperref driver's
% defaults. Encode internal destinations as PDF text; URI values are already
% complete, escaped ASCII PDF strings and must not be encoded a second time.
\ExplSyntaxOn
\str_new:N \l__pure_s_link_contents_str
\hook_gput_code:nnn {pdfannot/link/GoTo/before} {pure-s/descriptive-links}
  {
    \__hyp_text_pdfstring:eoN
      {Go~to~destination~\l_hyp_current_dest_name_tl}
      {\l__hyp_text_enc_info_print_tl}
      \l__pure_s_link_contents_str
    \pdfannot_dict_put:nne {link/GoTo}{Contents}{\l__pure_s_link_contents_str}
  }
\hook_gput_code:nnn {pdfannot/link/URI/before} {pure-s/descriptive-links}
  {
    \pdfannot_dict_put:nne {link/URI}{Contents}{\l__hyp_uri_tmpa_tl}
  }
\ExplSyntaxOff
